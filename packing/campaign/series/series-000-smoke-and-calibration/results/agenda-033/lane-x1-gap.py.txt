from __future__ import annotations
import json, math
from fractions import Fraction
from pathlib import Path
from devtools.owner_footprints import owner_branch_manifest, HALF_CORE
from devtools.screen_corner_dual_salvage import _base_polygon, screen_footprint
from devtools.transport_ceiling_family import transport
from sqpack.fractional.ceiling import CeilingCertificate

SRC = Path("/home/user/squares/packing/campaign/series/series-000-smoke-and-calibration/"
           "results/agenda-033/ceiling-family-191-50.json")
record = json.loads(SRC.read_text())
manifest = owner_branch_manifest()
directions = manifest.directions.directions
axes = frozenset((d.ux, d.uy) for d in directions) | frozenset((d.vx, d.vy) for d in directions)
moved = transport(record, Fraction(1), side=Fraction(96, 25))
fam = CeilingCertificate.from_record(moved)
by_id = {c.class_id: c for c in manifest.classes}

for cid in ("bottom-left:m1:j3", "bottom-left:m1:j4"):
    poly = _base_polygon(by_id[cid], "endpoint", directions)
    print(f"\n{cid}: footprint vertices {[(str(x), str(y)) for x, y in poly]}")
    print(f"  half-core arm length {HALF_CORE} = {float(HALF_CORE):.5f}")
    s = screen_footprint(fam, poly, class_id=cid, corner="bottom-left", kind="endpoint",
                         source_placements=fam.placements, direction_axes=axes)
    # each survivor carries an exact SAT gap along a (non-unit) axis; normalise it
    best = None
    for e in s.survivors:
        ax, ay = e.separation.axis
        norm = math.hypot(float(ax), float(ay))
        d = float(e.separation.gap) / norm
        if best is None or d < best[0]:
            best = (d, e.source_index, e.weight)
    print(f"  survivors {len(s.survivors)}, weight {s.total_weight}")
    print(f"  closest survivor: euclidean SAT gap {best[0]:.6f}, source index {best[1]}, "
          f"weight {best[2]}")
    p = fam.placements[best[1]]
    print(f"    that placement: half-tangent {p.half_tangent}, centre "
          f"({float(p.centre_x):.5f}, {float(p.centre_y):.5f})")
