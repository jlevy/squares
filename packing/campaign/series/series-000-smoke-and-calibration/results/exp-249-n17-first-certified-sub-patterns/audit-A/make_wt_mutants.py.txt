Each mutant is the committed module with one textual patch that must apply exactly once.
The repository file is read, never written.
"""

def main() -> int:
    source = SRC.read_text(encoding="utf-8")
    print("source sha256", hashlib.sha256(source.encode()).hexdigest())
    OUT.mkdir(exist_ok=True)
    names = sys.argv[1:] or list(MUTATIONS)
    for name in names:
        text = source
        for old, new in MUTATIONS[name]:
            assert text.count(old) == 1, (name, old)
            text = text.replace(old, new)
        (OUT / f"mut_{name}.py").write_text(text, encoding="utf-8")
        print("wrote", name)
    # The unmutated copy, for a baseline under the same harness.
    (OUT / "mut_none.py").write_text(source, encoding="utf-8")
    return 0


if __name__ == "__main__":
    sys.exit(main())
