def col(label: int, comp: str) -> int:
    return 3 * (label - 1) + {"x": 0, "y": 1, "w": 2}[comp]


def dot(a, b):
    return a[0] * b[0] + a[1] * b[1]


def rot(a):  # d/d omega of a rotating vector, at omega = 0
    return (-a[1], a[0])


def corners(aux, label, half):
    a1, a2 = (aux[name] for name in AXES[CLASS[label]])
    return [
        ((s1 * a1[0] + s2 * a2[0]) * half, (s1 * a1[1] + s2 * a2[1]) * half)
        for s1, s2 in ((1, 1), (1, -1), (-1, 1), (-1, -1))
    ]


def support(aux, label, n, half):
    return max(dot(n, q) for q in corners(aux, label, half))


def gap_row(n, sign, owner, other, c_owner, c_other, q, half):
    """Gradient of g = sign n.(c_other + R q - c_owner) - 1/2 at omega = 0 (51 columns)."""
    t = rot(n)
    row = [0] * 51
    row[col(other, "x")] += sign * n[0]
    row[col(other, "y")] += sign * n[1]
    row[col(owner, "x")] -= sign * n[0]
    row[col(owner, "y")] -= sign * n[1]
    d = (c_other[0] + q[0] - c_owner[0], c_other[1] + q[1] - c_owner[1])
    row[col(owner, "w")] += sign * dot(t, d)
    row[col(other, "w")] += sign * dot(n, rot(q))
    return row, sign * dot(n, d) - half


WALL_NORMAL = {"left": (1, 0), "right": (-1, 0), "bottom": (0, 1), "top": (0, -1)}


def wall_rows(aux, side, centres, label, wall, half, pick=None):
    """Rows of the tight corner gaps of `label` against `wall`; `pick` fixes the corners."""
    n = WALL_NORMAL[wall]
    c = centres[label - 1]
    offset = side if wall in ("right", "top") else 0
    made = []
    for index, q in enumerate(corners(aux, label, half)):
        g = dot(n, (c[0] + q[0], c[1] + q[1])) + offset
        if pick is None:
            tight = g == 0
        else:
            tight = index in pick
        if tight:
            row = [0] * 51
            row[col(label, "x")] += n[0]
            row[col(label, "y")] += n[1]
            row[col(label, "w")] += dot(n, rot(q))
            made.append((row[col(label, "w")], index, row, g))
    if pick is None:
        made.sort(key=lambda item: item[0])
    else:
        made = [item for index in pick for item in made if item[1] == index]
    return made


def sign_of(x):
    return 1 if x > 0 else -1


def face_rows(aux, centres, left, right, axis, half, pick=None):
    """The four elementary corner rows of a parallel face and the two Lemma 2 rows."""
    n = aux[axis]
    c_l, c_r = centres[left - 1], centres[right - 1]
    elementary = {}
    for owner, other in ((left, right), (right, left)):
        c_o, c_p = centres[owner - 1], centres[other - 1]
        sign = sign_of(dot(n, (c_p[0] - c_o[0], c_p[1] - c_o[1]))) if pick is None else pick[owner][0]
        cs = corners(aux, other, half)
        if pick is None:
            values = [sign * dot(n, q) for q in cs]
            low = min(values)
            near = [i for i, v in enumerate(values) if v == low]
            assert len(near) == 2, (left, right, near)
        else:
            near = pick[owner][1]
        rows = []
        for i in near:
            row, g = gap_row(n, sign, owner, other, c_o, c_p, cs[i], half)
            rows.append((i, row, g))
        elementary[owner] = (sign, near, rows)
    return elementary


def face_pick(elementary):
    return {owner: (sign, near) for owner, (sign, near, _) in elementary.items()}


def lemma2_rows(elementary, left, right):
    """E- and E+: elementary rows of one owner that are convex combinations of the other's."""
    found = []
    for owner, other in ((left, right), (right, left)):
        _, _, mine = elementary[owner]
        _, _, theirs = elementary[other]
        (_, y1, _), (_, y2, _) = theirs
        for idx, x, _ in mine:
            pivot = next((k for k in range(51) if y1[k] != y2[k]), None)
            assert pivot is not None
            alpha = (x[pivot] - y2[pivot]) / (y1[pivot] - y2[pivot])
            combo = [alpha * p + (1 - alpha) * q for p, q in zip(y1, y2)]
            if combo == x and 0 <= alpha <= 1:
                if x not in [f[0] for f in found]:
                    found.append((x, owner, alpha, idx))
    assert len(found) == 2, [(o, a) for _, o, a, _ in found]
    found.sort(key=lambda item: -item[0][col(left, "w")])  # E- has the larger omega_left
    return found


def nonparallel_row(aux, centres, left, right, axis, half, pick=None):
    n = aux[axis]
    owner = left if AXIS_CLASS[axis] == CLASS[left] else right
    assert AXIS_CLASS[axis] == CLASS[owner] and AXIS_CLASS[axis] != CLASS[left if owner == right else right]
    other = right if owner == left else left
    c_o, c_p = centres[owner - 1], centres[other - 1]
    cs = corners(aux, other, half)
    if pick is None:
        sign = sign_of(dot(n, (c_p[0] - c_o[0], c_p[1] - c_o[1])))
        values = [sign * dot(n, q) for q in cs]
        low = min(values)
        near = [i for i, v in enumerate(values) if v == low]
        assert len(near) == 1, (left, right, near)
        index = near[0]
    else:
        sign, index = pick
    row, g = gap_row(n, sign, owner, other, c_o, c_p, cs[index], half)
    return row, g, owner, other, sign, index


def moved(centres, v, w):
    a, b, z = w
    c = list(centres)
    c[4] = (c[4][0] - a, c[4][1])
    c[10] = (c[10][0] - b * v[0], c[10][1] - b * v[1])
    c[12] = (c[12][0] + z * v[0], c[12][1] + z * v[1])
    return tuple(c)


def build_rows(aux, side, centres, half, picks=None):
    """All 58 rows in the frozen order, keyed as the instrument keys them, plus gaps."""
    rows = {}
    gaps = {}
    chosen = {}
    for label, wall in ANCHORS:
        made = wall_rows(aux, side, centres, label, wall, half, None if picks is None or ("wall", label, wall) not in picks else picks["wall", label, wall])
        chosen["wall", label, wall] = [item[1] for item in made]
        for variant, (_, _, row, g) in enumerate(made):
            rows["wall", label, wall, variant] = row
            gaps["wall", label, wall, variant] = g
    for left, right, axis, _ in CONTACTS:
        parallel = CLASS[left] == CLASS[right]
        if parallel:
            if picks is None or ("pair", left, right) not in picks:
                elementary = face_rows(aux, centres, left, right, axis, half)
                pair_rows = lemma2_rows(elementary, left, right)
                chosen["pair", left, right] = (face_pick(elementary), [(o, i) for _, o, _, i in pair_rows])
                for variant, (row, owner, alpha, idx) in enumerate(pair_rows):
                    rows["pair", left, right, variant] = row
            else:
                pick, pattern = picks["pair", left, right]
                elementary = face_rows(aux, centres, left, right, axis, half, pick)
                for variant, (owner, idx) in enumerate(pattern):
                    rows["pair", left, right, variant] = next(r for i, r, _ in elementary[owner][2] if i == idx)
            gaps["pair", left, right, 0] = tuple(g for o in elementary.values() for _, _, g in o[2])
        else:
            row, g, owner, other, sign, index = nonparallel_row(aux, centres, left, right, axis, half, None if picks is None or ("pair", left, right) not in picks else picks["pair", left, right])
            chosen["pair", left, right] = (sign, index)
            rows["pair", left, right, 0] = row
            gaps["pair", left, right, 0] = g
    assert len(rows) == 58
    return rows, gaps, chosen


def names_and_lifts(u):
    names, lifts = [], []
    for label in range(1, 18):
        if label == 6:
            continue
        if label in (11, 13):
            names.append(f"u{label}")
            lifts.append({col(label, "x"): u[0], col(label, "y"): u[1]})
        elif label != 5:
            names.append(f"xi{label}")
            lifts.append({col(label, "x"): 1})
        if label not in (11, 13):
            names.append(f"eta{label}")
            lifts.append({col(label, "y"): 1})
        names.append(f"omega{label}")
        lifts.append({col(label, "w"): 1})
    assert len(names) == 45
    return names, lifts


def project(rows, keys, lifts):
    return [[sum(row[p] * c for p, c in lift.items()) for lift in lifts] for key in keys for row in (rows[key],)]


def sqrt_up(value):
    scaled = math.ceil(value * SCALE * SCALE)
    root = math.isqrt(scaled)
    if root * root < scaled:
        root += 1
    upper = Q(root, SCALE)
    assert upper * upper >= value
    return upper


def to_iv(q: Q, margin):
    v = mpmath.mpf(q.numerator) / mpmath.mpf(q.denominator)
    return iv.mpf([v - margin, v + margin])


def iv_to_q_upper(x) -> Q:
    """The exact rational value of an interval's upper endpoint."""
    return Q(*mpmath.libmp.to_rational(x._mpi_[1]))


def iv_to_q_lower(x) -> Q:
    return Q(*mpmath.libmp.to_rational(x._mpi_[0]))

                for sign in (1, -1):
                    c_o, c_p = centres[owner - 1], centres[other - 1]
                    base_margin = sign * dot(n, (c_p[0] - c_o[0], c_p[1] - c_o[1])) - HALF - support(aux, other, n, HALF)
                    identity = abs(base_margin) < Q(1, 10**30)
                    count += 1
                    if identity or (left, right) in DROPPED_PAIRS:
                        continue
                    assert base_margin < 0
                    cs = corners(aux, other, HALF)
                    pick = min(range(4), key=lambda i: sign * dot(n, cs[i]))
                    Kv = pair_K(owner, other)[0]
                    worst_margin = None
                    for vert in vertices:
                        c = moved(centres, v, vert)
                        row, g = gap_row(n, sign, owner, other, c[owner - 1], c[other - 1], cs[pick], HALF)
                        lin = sum(abs(sum(row[p] * cf for p, cf in lift.items())) * RADIUS for lift in lifts)
                        m = g + lin + Kv / 2
                        worst_margin = m if worst_margin is None else max(worst_margin, m)
                    option_rows.append((worst_margin, f"{left}/{right} owner {owner} axis {axis} sign {sign}", float(base_margin)))
    option_rows.sort(key=lambda x: -x[0])
    report["c6_options_total"] = count
    report["c6_unavailable_checked"] = len(option_rows)
    report["c6_all_negative"] = all(m < 0 for m, _, _ in option_rows)
    report["c6_least_negative"] = [(float(m), lab, bm) for m, lab, bm in option_rows[:6]]

    # 6. Physical slider domain of the 16-square family (float SAT), with and without 6.
    def polys(w, with6):
        c = moved(centres, v, w)
        out = {}
        for label in range(1, 18):
            if label == 6 and not with6:
                continue
            out[label] = [(float(c[label - 1][0] + q[0]), float(c[label - 1][1] + q[1])) for q in corners(aux, label, HALF)]
        return out

    def overlap(P, Qp, tol=1e-9):
        axes = []
        for poly in (P, Qp):
            for i in range(4):
                for k in range(i + 1, 4):
                    dx, dy = poly[k][0] - poly[i][0], poly[k][1] - poly[i][1]
                    L = math.hypot(dx, dy)
                    if abs(L - 1) < 1e-9:
                        axes.append((-dy, dx))
        for ax in axes:
            pa = [ax[0] * x + ax[1] * y for x, y in P]
            pb = [ax[0] * x + ax[1] * y for x, y in Qp]
            if min(pa) >= max(pb) - tol or min(pb) >= max(pa) - tol:
                return False
        return True

    def violations(w, with6):
        P = polys(w, with6)
        S = float(side)
        bad = []
        for label, poly in P.items():
            for x, y in poly:
                if x < -1e-9 or y < -1e-9 or x > S + 1e-9 or y > S + 1e-9:
                    bad.append(f"wall:{label}")
                    break
        for a1, b1 in itertools.combinations(P, 2):
            if overlap(P[a1], P[b1]):
                bad.append(f"{a1}/{b1}")
        return bad

    def feasible(w, with6):
        return not violations(w, with6)

    def scan(index, direction, with6, step=Q(1, 2000), limit=Q(3)):
        w = [Q(0)] * 3
        last = Q(0)
        value = Q(0)
        blockers = []
        while value <= limit:
            w[index] = direction * value
            blockers = violations(tuple(w), with6)
            if blockers:
                break
            last = value
            value += step
        return float(direction * last), blockers

    domain = {}
    for idx, name in enumerate("abz"):
        for with6 in (False, True):
            lo_v, lo_b = scan(idx, -1, with6)
            hi_v, hi_b = scan(idx, 1, with6)
            domain[f"{name}{'_with6' if with6 else '_no6'}"] = {
                "min": lo_v, "min_blocked_by": lo_b, "max": hi_v, "max_blocked_by": hi_b}
    report["physical_slider_domain_single_axis_steps_1/2000"] = domain
    report["family_feasible_at_origin_no6"] = feasible((Q(0), Q(0), Q(0)), False)
    report["family_feasible_at_origin_with6"] = feasible((Q(0), Q(0), Q(0)), True)
    report["box_vertices_feasible_no6"] = {str([str(x) for x in vert]): feasible(vert, False) for vert in vertices}
    (OUT / "recompute-output.json").write_text(json.dumps(report, indent=1))
    print(json.dumps({k: v for k, v in report.items() if k not in ("K_pairs", "chosen_corners", "box_vertices_feasible_no6")}, indent=1))
    print("box vertices feasible (no 6):", report["box_vertices_feasible_no6"])
