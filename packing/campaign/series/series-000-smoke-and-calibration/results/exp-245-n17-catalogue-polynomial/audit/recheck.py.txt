# ---- 3. exact evaluation and sign change -----------------------------------------------
def horner(coeffs: list[int] | list[Q], x: Q) -> Q:
    acc = Q(0)
    for c in coeffs:
        acc = acc * x + c
    return acc


def sgn(v: Q) -> int:
    return (v > 0) - (v < 0)

# ---- 4. root count: own Sturm sequence, and Descartes on a Moebius transform ----------
def poly_rem(a: list[Q], b: list[Q]) -> list[Q]:
    a = list(a)
    while len(a) >= len(b) and any(a):
        if a[0] == 0:
            a.pop(0)
            continue
        q = a[0] / b[0]
        for i in range(len(b)):
            a[i] -= q * b[i]
        a.pop(0)
    while a and a[0] == 0:
        a.pop(0)
    return a


def derivative(a: list[Q]) -> list[Q]:
    n = len(a) - 1
    return [c * (n - i) for i, c in enumerate(a[:-1])]


def sturm_chain(a: list[Q]) -> list[list[Q]]:
    chain = [a, derivative(a)]
    while True:
        r = poly_rem(chain[-2], chain[-1])
        if not r:
            return chain
        chain.append([-c for c in r])


def variations(chain: list[list[Q]], x: Q) -> int:
    signs = [sgn(horner(p, x)) for p in chain]
    signs = [s for s in signs if s]
    return sum(1 for i in range(1, len(signs)) if signs[i] != signs[i - 1])

# own distinct-degree test: f squarefree of degree n is irreducible mod p iff
# gcd(f, x^(p^i) - x) = 1 for every 1 <= i <= n//2.
def ddf_irreducible(coeffs: list[int], p: int) -> bool:
    n = len(coeffs) - 1
    inv = pow(coeffs[0], -1, p)
    f = [(c * inv) % p for c in reversed(coeffs)]  # lowest first, monic

    def rem(a: list[int]) -> list[int]:
        a = [v % p for v in a]
        while len(a) > n:
            if a[-1] == 0:
                a.pop()
                continue
            lead = a[-1]
            for i in range(n + 1):
                a[len(a) - 1 - n + i] = (a[len(a) - 1 - n + i] - lead * f[i]) % p
            a.pop()
        while a and a[-1] == 0:
            a.pop()
        return a

    def mul(a: list[int], b: list[int]) -> list[int]:
        out = [0] * (len(a) + len(b) - 1) if a and b else []
        for i, u in enumerate(a):
            for j, v in enumerate(b):
                out[i + j] = (out[i + j] + u * v) % p
        return rem(out)

    def powm(a: list[int], e: int) -> list[int]:
        r, base = [1], rem(a)
        while e:
            if e & 1:
                r = mul(r, base)
            base = mul(base, base)
            e >>= 1
        return r

    def gcd_(a: list[int], b: list[int]) -> list[int]:
        a, b = [v % p for v in a], [v % p for v in b]
        while b and any(b):
            while b and b[-1] == 0:
                b.pop()
            if not b:
                break
            # a mod b
            a = [v % p for v in a]
            while a and a[-1] == 0:
                a.pop()
            while len(a) >= len(b):
                lead = a[-1] * pow(b[-1], -1, p) % p
                sh = len(a) - len(b)
                for i, v in enumerate(b):
                    a[sh + i] = (a[sh + i] - lead * v) % p
                while a and a[-1] == 0:
                    a.pop()
            a, b = b, a
        while a and a[-1] == 0:
            a.pop()
        return a

    # squarefree check: gcd(f, f') = 1
    fprime = [(i * c) % p for i, c in enumerate(f)][1:]
    if len(gcd_(f, fprime)) != 1:
        return False
    xp = [0, 1]
    for i in range(1, n // 2 + 1):
        xp = powm(xp, p)
        h = list(xp) + [0, 0]
        h[1] = (h[1] - 1) % p
        if len(gcd_(f, h)) != 1:
            return False
    return True
