rec = json.loads((REPO / "packing/campaign/series/series-000-smoke-and-calibration/results/exp-245-n17-catalogue-polynomial/run-001/receipt.json").read_bytes())
C = rec["catalogue"]["coefficients"]
s = symbols("s")
def is_prime(n): return n > 1 and all(n % d for d in range(2, int(n**0.5) + 1))
irr = []
for p in (p for p in range(2, 1000) if is_prime(p) and 4775 % p):
    _, mf = Poly(C, s, modulus=p).factor_list()
    if sorted(f.degree() for f, m in mf for _ in range(m)) == [18]:
        irr.append(p)
print("primes < 1000 (not dividing 4775) with C irreducible mod p:", irr, "count", len(irr), "of", sum(1 for p in range(2,1000) if is_prime(p) and 4775 % p))

for p in irr[:4]:
    print(f"own DDF test mod {p}:", ns["ddf_irreducible"](C, p))
