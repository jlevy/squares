#!/usr/bin/env python3
"""Spike B: shared separation core for threshold atoms against a placement family.

Exact geometry (Fractions) where a decision is taken, floats where they only screen:
overlap graph by separating axes, pairwise intersection polygons by Sutherland-Hodgman,
representative points as vertex averages, containment matrices with a 1e-9 float margin
and Placement.contains on the ambiguous pairs, 2-of-3 charges over the whole family by
int8 count matrices, exact rational recheck of every candidate reported as violated.
Also the LP-side bookkeeping: atom orbits as columns (coefficient = number of orbit images
the row charges, cost = orbit size times floor(|S|/k)), computed per direction in floats
with the record's COVER_SLACK loosening (the LP sees at least the exact charge).
"""

from __future__ import annotations

import math
import time
from collections import Counter
from fractions import Fraction
from typing import Any

import numpy as np
from scipy import sparse

from sqpack.fractional.ceiling import CeilingCertificate, Placement, float_family
from sqpack.fractional.cutting import (
    COVER_SLACK,
    ExactRow,
    support_entries,
    symmetric_placements,
    tidy_family,
)
from sqpack.fractional.generate import direction_net
from sqpack.fractional.threshold import ThresholdAtom

MARGIN = 1e-9
Point = tuple[Fraction, Fraction]


def clip(poly, a, b, c):
    """Keep the part of the convex polygon with a x + b y <= c."""
    out = []
    n = len(poly)
    for i in range(n):
        p, q = poly[i], poly[(i + 1) % n]
        fp = a * p[0] + b * p[1] - c
        fq = a * q[0] + b * q[1] - c
        if fp <= 0:
            out.append(p)
        if (fp < 0 < fq) or (fq < 0 < fp):
            t = fp / (fp - fq)
            out.append((p[0] + t * (q[0] - p[0]), p[1] + t * (q[1] - p[1])))
    return out


def half_planes(p: Placement):
    ax, ay, u, bx, by, v = p.slabs()
    h = p.side / 2
    return ((ax, ay, u + h), (-ax, -ay, h - u), (bx, by, v + h), (-bx, -by, h - v))


def intersect(poly, p: Placement):
    for a, b, c in half_planes(p):
        poly = clip(poly, a, b, c)
        if not poly:
            return []
    return poly


def area(poly) -> Fraction:
    s = Fraction(0)
    for i in range(len(poly)):
        p, q = poly[i], poly[(i + 1) % len(poly)]
        s += p[0] * q[1] - q[0] * p[1]
    return abs(s) / 2


def overlap_exact(p: Placement, q: Placement, cp, cq) -> bool:
    """Closed squares meet iff no edge normal of either separates the corner sets."""
    for s, other in ((p, cq), (q, cp)):
        ax, ay, u, bx, by, v = s.slabs()
        h = s.side / 2
        for nx, ny, off in ((ax, ay, u), (bx, by, v)):
            vals = [nx * x + ny * y for x, y in other]
            if max(vals) < off - h or min(vals) > off + h:
                return False
    return True


def fold_deg(t: Fraction) -> float:
    a = math.degrees(2 * math.atan(float(t)))
    return a if a <= 45 else 90 - a


def angle_class(a: float) -> str:
    return "axis" if a <= 2.5 else ("29" if 25 <= a <= 35 else "other")


def family_from_dual(
    exact_rows: list[ExactRow],
    duals: np.ndarray,
    half_tangents: tuple[Fraction, ...],
    outer_side: Fraction,
    square_side: Fraction,
    n: int = 11,
    weight_denominator: int = 10**9,
) -> CeilingCertificate:
    """The LP dual as an exact D4-symmetric family (weights y/8 on the eight images)."""
    entries = support_entries(
        exact_rows, duals, half_tangents, support_cap=len(exact_rows), weight_denominator=weight_denominator
    )
    return CeilingCertificate(n, outer_side, square_side, half_tangents, symmetric_placements(entries, outer_side, square_side))


class FamilyGeometry:
    """Everything the separation needs about one family, built once."""

    def __init__(self, family: CeilingCertificate, log=print) -> None:
        t0 = time.perf_counter()
        # Merge identical placements first: `symmetric_placements` keeps the repeated
        # images of a placement on a mirror, and two identical squares would give two
        # edges the same representative point.
        merged: dict[tuple, Placement] = {}
        for q in family.placements:
            key = (q.half_tangent, q.centre_x, q.centre_y, q.side)
            if key in merged:
                old = merged[key]
                merged[key] = Placement(q.half_tangent, q.centre_x, q.centre_y, old.weight + q.weight, q.side)
            else:
                merged[key] = q
        family = CeilingCertificate(family.n, family.outer_side, family.square_side, family.half_tangents, tuple(merged.values()))
        self.family = tidy_family(family, 10**9)
        pl = self.family.placements
        self.pl = pl
        n = len(pl)
        self.n = n
        self.y = np.array([float(p.weight) for p in pl])
        self.yq = [p.weight for p in pl]
        self.corners = [p.corners() for p in pl]
        self.angles = [fold_deg(p.half_tangent) for p in pl]
        self.classes = [angle_class(a) for a in self.angles]
        cx = np.array([float(p.centre_x) for p in pl])
        cy = np.array([float(p.centre_y) for p in pl])
        side = float(pl[0].side)
        d2 = (cx[:, None] - cx[None, :]) ** 2 + (cy[:, None] - cy[None, :]) ** 2
        cand = np.argwhere(np.triu(d2 <= (side * math.sqrt(2) + 1e-6) ** 2, 1))
        self.edges = [
            (int(i), int(j)) for i, j in cand if overlap_exact(pl[i], pl[j], self.corners[i], self.corners[j])
        ]
        self.adj = [set() for _ in range(n)]
        for i, j in self.edges:
            self.adj[i].add(j)
            self.adj[j].add(i)
        self.edge_index = {e: k for k, e in enumerate(self.edges)}
        self.reps: list[Point] = []
        for i, j in self.edges:
            poly = intersect(list(self.corners[i]), pl[j])
            m = len(poly)
            self.reps.append((sum(p[0] for p in poly) / m, sum(p[1] for p in poly) / m))
        self.normals, self.offsets, self.halves, _ = float_family(self.family)
        self.C, self.ambiguous = containment(self.reps, pl, self.normals, self.offsets, self.halves)
        self.Ci = self.C.astype(np.int8)
        log(
            f"  geometry: {n} placements, {len(self.edges)} edges, mean degree "
            f"{2*len(self.edges)/max(n,1):.1f}, {self.ambiguous} containments decided exactly, "
            f"{time.perf_counter()-t0:.1f}s"
        )

    def triangles(self) -> np.ndarray:
        tri = []
        for i, j in self.edges:
            for k in self.adj[i] & self.adj[j]:
                if k > j:
                    tri.append((i, j, k))
        return np.array(tri, dtype=np.int64).reshape(-1, 3)

    def charges_2of3(self, tri: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        e1 = np.array([self.edge_index[(i, j)] for i, j, k in tri], dtype=np.int64)
        e2 = np.array([self.edge_index[(j, k)] for i, j, k in tri], dtype=np.int64)
        e3 = np.array([self.edge_index[(i, k)] for i, j, k in tri], dtype=np.int64)
        charges = np.zeros(len(tri))
        step = 20000
        for s in range(0, len(tri), step):
            cnt = self.Ci[e1[s : s + step]] + self.Ci[e2[s : s + step]] + self.Ci[e3[s : s + step]]
            charges[s : s + step] = (cnt >= 2) @ self.y
        return charges, e1, e2, e3

    def exact_charge(self, point_ids: list[int], threshold: int) -> Fraction:
        cnt = sum(self.Ci[e] for e in point_ids)
        members = np.flatnonzero(cnt >= threshold)
        return sum((self.yq[m] for m in members), start=Fraction(0))

    def vertex_candidates(self, *, total_cap: int = 60000, log=print) -> None:
        """Arrangement vertices as extra candidate points, deepest first, exact points.

        Sets ``self.vpts`` (exact points), ``self.VC`` (containment, exact where ambiguous)
        and ``self.vdepth``. For a large family only the ``total_cap`` deepest vertices
        (float depth) are kept and rebuilt exactly from their two lines.
        """
        t0 = time.perf_counter()
        from sqpack.fractional.ceiling import arrangement_lines, exact_intersection, float_family, loose_membership
        from sqpack.fractional.cutting import float_vertices
        lines = arrangement_lines(self.family)
        points, pairs, cache = float_vertices(self.family, lines)
        normals, offsets, halves, weights = float_family(self.family)
        depth = np.zeros(points.shape[0])
        chunk = max(1, 2_000_000 // max(1, self.n))
        for s in range(0, points.shape[0], chunk):
            block = points[s : s + chunk]
            depth[s : s + chunk] = loose_membership(block, normals, offsets, halves, margin=2e-6).astype(float) @ weights
        order = np.argsort(-depth, kind="stable")[:total_cap]
        side = self.family.outer_side
        exact_pts = []
        for idx in order:
            e = cache.get(int(idx))
            if e is None:
                e = exact_intersection(lines[int(pairs[idx, 0])], lines[int(pairs[idx, 1])])
            if e is None or not (0 <= e[0] <= side and 0 <= e[1] <= side):
                continue
            exact_pts.append(e)
        seen = set(); uniq = []
        for e in exact_pts:
            if e not in seen:
                seen.add(e); uniq.append(e)
        self.vpts = uniq
        self.VC, amb = containment(uniq, self.pl, self.normals, self.offsets, self.halves)
        self.VCi = self.VC.astype(np.int8)
        self.vdepth = self.VC @ self.y
        log(f"  vertex candidates: {points.shape[0]} vertices, kept {len(uniq)} deepest (max depth {self.vdepth.max():.4f}), {amb} decided exactly, {time.perf_counter()-t0:.1f}s")

    def interior_candidates(self, e: int, *, cap: int, pull: Fraction = Fraction(1, 1000)):
        """Interior candidate points for edge ``e``: the deepest ``cap`` arrangement vertices
        inside both placements, each pulled towards the pair polygon's centroid by ``pull``
        (so it lies strictly inside both), plus the centroid; with their exact containment."""
        if not hasattr(self, "_icache"):
            self._icache = {}
        key = (e, cap, pull)
        if key in self._icache:
            return self._icache[key]
        i, j = self.edges[e]
        idx = np.flatnonzero(self.VC[:, i] & self.VC[:, j])
        if idx.size > cap:
            idx = idx[np.argsort(-self.vdepth[idx], kind="stable")[:cap]]
        cx, cy = self.reps[e]
        pts = [(x + pull * (cx - x), y + pull * (cy - y)) for x, y in (self.vpts[t] for t in idx)] + [(cx, cy)]
        C, _ = containment(pts, self.pl, self.normals, self.offsets, self.halves)
        self._icache[key] = (pts, C.astype(np.int8))
        return self._icache[key]

    def refine_triangle(self, i: int, j: int, k: int, *, cap: int = 24) -> tuple[Fraction, tuple[Point, Point, Point]] | None:
        """The best 2-of-3 atom on triangle (i, j, k) with interior vertex candidates
        (`interior_candidates`) in each pairwise intersection, all combinations."""
        eij, ejk, eik = self.edge_index[(i, j)], self.edge_index[(j, k)], self.edge_index[(i, k)]
        ptsA, MA = self.interior_candidates(eij, cap=cap)
        ptsB, MB = self.interior_candidates(ejk, cap=cap)
        ptsC, MC = self.interior_candidates(eik, cap=cap)
        cnt = MA[:, None, None, :] + MB[None, :, None, :] + MC[None, None, :, :]
        charged = (cnt >= 2) @ self.y
        a, b, c = np.unravel_index(int(charged.argmax()), charged.shape)
        pts = (ptsA[a], ptsB[b], ptsC[c])
        if len(set(pts)) < 3:
            return None
        members = np.flatnonzero(cnt[a, b, c] >= 2)
        exact = sum((self.yq[m] for m in members), start=Fraction(0))
        return exact, pts

    def violated_2of3(self, *, outer_side: Fraction, min_violation: float = 0.0, log=print, vertex_cap: int = 0, refine_above: float = 0.85) -> list[dict[str, Any]]:
        """Every 2-of-3 atom on triangle edge representatives whose exact charge exceeds 1,
        one per D4 orbit (the heaviest), sorted by violation. With ``vertex_cap > 0`` the
        points of every triangle whose centroid charge reaches ``refine_above`` are re-chosen
        among the deepest ``vertex_cap`` arrangement vertices inside each pairwise
        intersection (all combinations), which is what separates an integral-looking dual."""
        t0 = time.perf_counter()
        tri = self.triangles()
        charges, e1, e2, e3 = self.charges_2of3(tri)
        order = np.argsort(-charges, kind="stable")
        found: dict[tuple, dict[str, Any]] = {}
        checked = 0
        refined = 0
        if vertex_cap > 0 and not hasattr(self, "VC"):
            self.vertex_candidates(log=log)
        for idx in order:
            floor = (1 + min_violation - 1e-7) if vertex_cap == 0 else min(refine_above, 1 + min_violation - 1e-7)
            if charges[idx] < floor:
                break
            checked += 1
            pts = (self.reps[e1[idx]], self.reps[e2[idx]], self.reps[e3[idx]])
            if len(set(pts)) < 3:
                continue
            exact = self.exact_charge([e1[idx], e2[idx], e3[idx]], 2)
            if vertex_cap > 0:
                i, j, k = (int(v) for v in tri[idx])
                better = self.refine_triangle(i, j, k, cap=vertex_cap)
                refined += 1
                if better is not None and better[0] > exact:
                    exact, pts = better
            if exact <= 1 + Fraction(min_violation).limit_denominator(10**9):
                continue
            atom = ThresholdAtom(pts, 2, Fraction(1))
            key = min(image.key for image in atom.images(outer_side))
            if key in found and found[key]["charge"] >= exact:
                continue
            i, j, k = (int(v) for v in tri[idx])
            found[key] = {
                "kind": "2-of-3",
                "threshold": 2,
                "points": pts,
                "charge": exact,
                "budget": 1,
                "triangle": [i, j, k],
                "classes": [self.classes[i], self.classes[j], self.classes[k]],
                "angles": [self.angles[i], self.angles[j], self.angles[k]],
            }
        atoms = sorted(found.values(), key=lambda a: -a["charge"])
        log(
            f"  separation: {len(tri)} triangles, max centroid charge {charges.max() if len(tri) else 0:.6f}, "
            f"{int((charges > 1).sum())} violated (float), {checked} rechecked exactly, {refined} vertex-refined, "
            f"{len(atoms)} violated orbits (max charge {float(atoms[0]['charge']) if atoms else 0:.6f}), {time.perf_counter()-t0:.1f}s"
        )
        return atoms


def containment(points, pl, normals, offsets, halves):
    """Exact containment matrix [points x placements]: floats screen, Fractions decide."""
    P = np.array([[float(x), float(y)] for x, y in points])
    first = np.abs(P @ normals[:, 0, :].T - offsets[None, :, 0])
    second = np.abs(P @ normals[:, 1, :].T - offsets[None, :, 1])
    loose = (first <= halves[None, :] + MARGIN) & (second <= halves[None, :] + MARGIN)
    strict = (first <= halves[None, :] - MARGIN) & (second <= halves[None, :] - MARGIN)
    result = strict.copy()
    ambiguous = np.argwhere(loose & ~strict)
    for a, b in ambiguous:
        result[a, b] = pl[b].contains(points[a][0], points[a][1])
    return result, len(ambiguous)


def atom_columns(
    orbits: list[tuple[ThresholdAtom, ...]],
    exact_rows: list[ExactRow],
    half_tangents: tuple[Fraction, ...],
    square_side: Fraction,
    *,
    slack: float = COVER_SLACK,
) -> tuple[sparse.csr_matrix, np.ndarray]:
    """Coefficient matrix [rows x orbits] and cost vector for threshold-atom orbit columns.

    Coefficient: the number of images in the orbit that the row's placement charges
    (contains at least k of the image's points), with the LP's loosening. Cost: orbit size
    times floor(|S|/k), the orbit's budget per unit weight.
    """
    directions = direction_net(half_tangents)
    n_rows = len(exact_rows)
    images: list[ThresholdAtom] = []
    orbit_of: list[int] = []
    for o, orbit in enumerate(orbits):
        for image in orbit:
            images.append(image)
            orbit_of.append(o)
    costs = np.array([len(orbit) * (orbit[0].size // orbit[0].threshold) for orbit in orbits], dtype=float)
    if not images or n_rows == 0:
        return sparse.csr_matrix((n_rows, len(orbits))), costs
    max_size = max(im.size for im in images)
    pts = np.full((len(images), max_size, 2), np.nan)
    ks = np.array([im.threshold for im in images])
    for a, im in enumerate(images):
        for b, (x, y) in enumerate(im.points):
            pts[a, b] = (float(x), float(y))
    orbit_of_arr = np.array(orbit_of)
    row_dirs = np.array([r[0] for r in exact_rows])
    xs = np.array([float(r[1]) for r in exact_rows])
    ys = np.array([float(r[2]) for r in exact_rows])
    half = float(square_side) / 2 + slack
    rows_out: list[np.ndarray] = []
    cols_out: list[np.ndarray] = []
    vals_out: list[np.ndarray] = []
    for d, direction in enumerate(directions):
        which = np.flatnonzero(row_dirs == d)
        if which.size == 0:
            continue
        cos, sin = float(direction.ux), float(direction.uy)
        pu = pts[..., 0] * cos + pts[..., 1] * sin  # (images, size)
        pv = -pts[..., 0] * sin + pts[..., 1] * cos
        cu = xs[which] * cos + ys[which] * sin
        cv = -xs[which] * sin + ys[which] * cos
        inside = (np.abs(pu[None, :, :] - cu[:, None, None]) <= half) & (
            np.abs(pv[None, :, :] - cv[:, None, None]) <= half
        )
        count = inside.sum(axis=2)  # (rows_d, images)
        charged = count >= ks[None, :]
        r, im = np.nonzero(charged)
        if r.size == 0:
            continue
        rows_out.append(which[r])
        cols_out.append(orbit_of_arr[im])
        vals_out.append(np.ones(r.size))
    if not rows_out:
        return sparse.csr_matrix((n_rows, len(orbits))), costs
    matrix = sparse.coo_matrix(
        (np.concatenate(vals_out), (np.concatenate(rows_out), np.concatenate(cols_out))),
        shape=(n_rows, len(orbits)),
    ).tocsr()
    matrix.sum_duplicates()
    return matrix, costs


def class_summary(atoms: list[dict[str, Any]]) -> dict[str, int]:
    return dict(Counter("-".join(sorted(a["classes"])) for a in atoms))
