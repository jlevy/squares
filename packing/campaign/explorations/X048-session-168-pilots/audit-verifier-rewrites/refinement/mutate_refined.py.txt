"""Lane R6's doctored refined certificates against the adaptive-rows kernel verifier.

Run from packing/ with PYTHONPATH=. :

    .venv/bin/python3 <this file> CERT_DIR CELLS_JSON --module VERIFIER [--only NAME] [--json OUT]

CERT_DIR holds one refined seed and node (lane K2's `k2-refined/{blind,wall}`), CELLS_JSON
its cells. Each mutation edits a deep copy, re-saves it with the checker tool's saver, and
runs the verifier's `verify_objects`; the unmutated objects must be accepted. Beyond lane
P2's nine doctored refinements (gap, overlap, short, wrong parents, empty row, grandparent,
partner, lost regions), these target the start of the partition, reversed and self-citing
rows, a child crossing its parent's upper end, partner covers of the wrong length or
order, and cores strict on half of a refined row only, which only a check over the row's
whole interval refuses.
"""

from __future__ import annotations

import argparse
import copy
import importlib.util
import json
import sys
import tempfile
import time
from collections.abc import Callable
from fractions import Fraction as Q
from pathlib import Path
from types import ModuleType
from typing import Any

from devtools.check_n17_subpattern import load_certificate, save_certificate

Point = tuple[Q, Q]
Mutation = Callable[[dict[str, Any], dict[str, Any], ModuleType], str]


def load(path: Path, name: str) -> ModuleType:
    spec = importlib.util.spec_from_file_location(name, path)
    assert spec is not None and spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


def dec(polygon: Any) -> list[Point]:
    return [(Q(x), Q(y)) for x, y in polygon]


def enc(polygon: list[Point]) -> list[list[str]]:
    return [[str(x), str(y)] for x, y in polygon]


def refined_rows(node: dict[str, Any]) -> list[tuple[int, int, dict[str, Any], tuple[Q, Q]]]:
    """(step, row index, row, parent interval) for rows narrower than their predecessor."""
    out = []
    accepted: dict[int, dict[str, tuple[Q, Q]]] = {}
    for si, step in enumerate(node["steps"]):
        owner = step["owner"]
        for ri, row in enumerate(step["rows"]):
            key = json.dumps(row["prior_reference"], sort_keys=True)
            lo, hi = Q(row["interval"][0]), Q(row["interval"][1])
            parent = accepted.get(owner, {}).get(key)
            if parent is not None and parent != (lo, hi):
                out.append((si, ri, row, parent))
        accepted[owner] = {
            json.dumps(row["reference"], sort_keys=True): (Q(row["interval"][0]), Q(row["interval"][1]))
            for row in step["rows"]
        }
    return out


def first_split(node: dict[str, Any]) -> tuple[int, int]:
    """The first step and row index where a row shares its predecessor with the next."""
    for si, step in enumerate(node["steps"]):
        rows = step["rows"]
        for ri in range(len(rows) - 1):
            if rows[ri]["prior_reference"] == rows[ri + 1]["prior_reference"]:
                return si, ri
    raise RuntimeError("no split")


# -- partition and citation ------------------------------------------------------------


def start_above_zero(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    """The closing step's first row starts at 2^-20: the angles below are uncovered."""
    node["steps"][-1]["rows"][0]["interval"][0] = str(Q(1, 2**20))
    return "interval gap"


def end_below_one(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    node["steps"][-1]["rows"][-1]["interval"][1] = str(1 - Q(1, 2**20))
    return "do not reach the end"


def reversed_interval(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    si, ri = first_split(node)
    row = node["steps"][si]["rows"][ri]
    row["interval"] = [row["interval"][1], row["interval"][0]]
    return "interval"


def self_citation(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    row = node["steps"][-1]["rows"][0]
    row["prior_reference"] = copy.deepcopy(row["reference"])
    return "not an accepted row"


def child_crosses_parent(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    """The second child of the first split ends 2^-30 past its parent, and the row after
    starts there, so the partition holds and only containment can refuse."""
    si, ri = first_split(node)
    rows = node["steps"][si]["rows"]
    if ri + 2 >= len(rows):
        raise RuntimeError("the split is at the end of the step")
    shift = Q(1, 2**30)
    rows[ri + 1]["interval"][1] = str(Q(rows[ri + 1]["interval"][1]) + shift)
    rows[ri + 2]["interval"][0] = str(Q(rows[ri + 2]["interval"][0]) + shift)
    return "escapes its predecessor"


def whole_parent_twice(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    """Both children of the first split claim the parent's whole interval."""
    si, ri = first_split(node)
    rows = node["steps"][si]["rows"]
    lo, hi = rows[ri]["interval"][0], rows[ri + 1]["interval"][1]
    rows[ri]["interval"] = [lo, hi]
    rows[ri + 1]["interval"] = [lo, hi]
    return "interval overlap"


# -- partner covers of variable length ---------------------------------------------------


def refined_cover_step(node: dict[str, Any]) -> tuple[int, str]:
    """A step whose partner cover has more items than the seed's bins."""
    bins = len(node["initial"]["cell_references"][next(iter(node["initial"]["cell_references"]))])
    for si, step in enumerate(node["steps"]):
        for pj, items in step["prior_partner_pose_covers"].items():
            if len(items) > bins:
                return si, pj
    raise RuntimeError("no partner cover longer than the seed grid")


def cover_missing_last_item(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    si, pj = refined_cover_step(node)
    node["steps"][si]["prior_partner_pose_covers"][pj].pop()
    return "cover length"


def cover_items_swapped(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    si, pj = refined_cover_step(node)
    items = node["steps"][si]["prior_partner_pose_covers"][pj]
    items[0], items[1] = items[1], items[0]
    return "reference"


def cover_item_interval_widened(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
    """A refined partner row's cover item claims the parent's whole interval."""
    si, pj = refined_cover_step(node)
    items = node["steps"][si]["prior_partner_pose_covers"][pj]
    items[0]["interval"][1] = str(Q(items[0]["interval"][1]) * 2)
    return "interval"


# -- cores strict on half a row ------------------------------------------------------------


def scale_core(core: list[Point], factor: Q) -> list[Point]:
    return [(x * factor, y * factor) for x, y in core]


def largest_strict_factor(v: ModuleType, core: list[Point], lo: Q, hi: Q) -> Q:
    """sup of the factors under which the scaled core is strict over [lo, hi], by
    bisection to 2^-48 (strictness is monotone in the factor)."""
    low, high = Q(1), Q(8)
    assert v.core_strict(scale_core(core, low), lo, hi)
    assert not v.core_strict(scale_core(core, high), lo, hi)
    for _ in range(48):
        mid = (low + high) / 2
        if v.core_strict(scale_core(core, mid), lo, hi):
            low = mid
        else:
            high = mid
    return low


def recompute_planes(row: dict[str, Any]) -> None:
    core = dec(row["core_vertices"])
    vertices = [p for polygon in row["residual_polygons"] for p in dec(polygon)]
    if not vertices:
        return
    planes = []
    n = len(core)
    for k in range(n):
        p, q = core[k], core[(k + 1) % n]
        a, b = q[1] - p[1], p[0] - q[0]
        least = min(a * x + b * y for x, y in vertices)
        planes.append({"normal": [str(a), str(b)], "upper": str(a * p[0] + b * p[1] + least)})
    row["common_core_halfplanes"] = planes


def core_strict_on_a_half_only(which: str) -> Mutation:
    """A refined row's core scaled to be strict over one half of its interval but not the
    other: a verifier checking that half alone would accept it."""

    def mutate(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
        for si, ri, row, _parent in reversed(refined_rows(node)):
            if not row["core_vertices"]:
                continue
            core = v.hull(dec(row["core_vertices"]))
            lo, hi = Q(row["interval"][0]), Q(row["interval"][1])
            mid = (lo + hi) / 2
            half = (lo, mid) if which == "lower" else (mid, hi)
            f_half = largest_strict_factor(v, core, *half)
            f_full = largest_strict_factor(v, core, lo, hi)
            if f_half <= f_full + Q(1, 2**40):
                continue
            factor = (f_half + f_full) / 2
            scaled = scale_core(core, factor)
            assert v.core_strict(scaled, *half) and not v.core_strict(scaled, lo, hi)
            row["core_vertices"] = enc(scaled)
            recompute_planes(row)
            return "core not strict"
        raise RuntimeError(f"no refined row whose core strictness binds outside its {which} half")

    return mutate


def partner_core_strict_on_a_half_only(which: str) -> Mutation:
    """A refined partner row's published core scaled likewise."""

    def mutate(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
        bins = len(node["initial"]["cell_references"][next(iter(node["initial"]["cell_references"]))])
        for si in range(len(node["steps"]) - 1, -1, -1):
            step = node["steps"][si]
            for pj, items in step["prior_partner_pose_covers"].items():
                if len(items) <= bins:
                    continue
                for item in items:
                    if not item["core"]:
                        continue
                    core = v.hull(dec(item["core"]))
                    lo, hi = Q(item["interval"][0]), Q(item["interval"][1])
                    mid = (lo + hi) / 2
                    half = (lo, mid) if which == "lower" else (mid, hi)
                    f_half = largest_strict_factor(v, core, *half)
                    f_full = largest_strict_factor(v, core, lo, hi)
                    if f_half <= f_full + Q(1, 2**40):
                        continue
                    factor = (f_half + f_full) / 2
                    scaled = scale_core(core, factor)
                    assert v.core_strict(scaled, *half) and not v.core_strict(scaled, lo, hi)
                    item["core"] = enc(scaled)
                    return "core not strict"
        raise RuntimeError(f"no refined partner row whose core strictness binds outside its {which} half")

    return mutate


def point_strict_on_one_half(v: ModuleType, lo: Q, hi: Q, which: str) -> Point | None:
    """A point strictly inside the unit square at every half-angle of one half of
    `[lo, hi]` but not at every half-angle of the whole row, by a search over directions
    and radii near the square's boundary. None when the row admits no such point."""
    import math

    mid = (lo + hi) / 2
    half = (lo, mid) if which == "lower" else (mid, hi)
    for k in range(720):
        phi = math.pi * k / 360
        direction = (Q(round(math.cos(phi) * 2**16), 2**16), Q(round(math.sin(phi) * 2**16), 2**16))
        for step in range(0, 160):
            r = Q(45, 100) + Q(step, 512)
            p = (r * direction[0], r * direction[1])
            if v.core_strict([p], *half) and not v.core_strict([p], lo, hi):
                return p
    return None


def core_grown_by_a_half_strict_point(which: str) -> Mutation:
    """A refined row's core grown by one point that is strict over one half of the row's
    interval only; the core still contains the old one, so every other obligation of the
    row holds and only strictness over the whole interval refuses it. Steps with an empty
    kernel are preferred, so no kernel point meets the recomputed planes."""

    def mutate(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
        candidates = sorted(
            refined_rows(node), key=lambda t: (bool(node["steps"][t[0]]["common_owned_kernel"]), -t[0])
        )
        for si, ri, row, _parent in candidates:
            if not row["core_vertices"]:
                continue
            lo, hi = Q(row["interval"][0]), Q(row["interval"][1])
            p = point_strict_on_one_half(v, lo, hi, which)
            if p is None:
                continue
            core = v.hull(dec(row["core_vertices"]))
            grown = v.hull([*core, p])
            mid = (lo + hi) / 2
            half = (lo, mid) if which == "lower" else (mid, hi)
            assert v.core_strict(grown, *half) and not v.core_strict(grown, lo, hi)
            row["core_vertices"] = enc(grown)
            recompute_planes(row)
            return "core not strict"
        raise RuntimeError(f"no refined row admits a point strict on its {which} half only")

    return mutate


def partner_core_grown_by_a_half_strict_point(which: str) -> Mutation:
    """A refined partner row's published core grown likewise."""

    def mutate(seed: dict[str, Any], node: dict[str, Any], v: ModuleType) -> str:
        bins = len(node["initial"]["cell_references"][next(iter(node["initial"]["cell_references"]))])
        for si in range(len(node["steps"]) - 1, -1, -1):

                    if p is None:

    return mutate


MUTATIONS: dict[str, Mutation] = {
    "owner-core-grown-by-a-point-strict-on-lower-half-only": core_grown_by_a_half_strict_point("lower"),
    "owner-core-grown-by-a-point-strict-on-upper-half-only": core_grown_by_a_half_strict_point("upper"),
    "partner-core-grown-by-a-point-strict-on-lower-half-only": partner_core_grown_by_a_half_strict_point("lower"),
    "partner-core-grown-by-a-point-strict-on-upper-half-only": partner_core_grown_by_a_half_strict_point("upper"),
    "partition-starts-above-zero": start_above_zero,
    "partition-ends-below-one": end_below_one,
    "row-interval-reversed": reversed_interval,
    "row-cites-itself": self_citation,
    "child-crosses-parent-upper-end": child_crosses_parent,
    "both-children-claim-the-parent": whole_parent_twice,
    "partner-cover-missing-last-item": cover_missing_last_item,
    "partner-cover-items-swapped": cover_items_swapped,
    "partner-cover-item-interval-widened": cover_item_interval_widened,
    "owner-core-strict-on-lower-half-only": core_strict_on_a_half_only("lower"),
    "owner-core-strict-on-upper-half-only": core_strict_on_a_half_only("upper"),
    "partner-core-strict-on-lower-half-only": partner_core_strict_on_a_half_only("lower"),
    "partner-core-strict-on-upper-half-only": partner_core_strict_on_a_half_only("upper"),
}


def run(v: ModuleType, directory: Path, cells: Any) -> tuple[str, str, float]:
    clock = time.perf_counter()
    try:
        result = v.verify_objects(directory, cells)
    except v.VerificationError as failure:
        return "REFUSED", str(failure), time.perf_counter() - clock
    except (KeyError, TypeError, ValueError, IndexError, AttributeError, ZeroDivisionError) as failure:
        return "MALFORMED", repr(failure), time.perf_counter() - clock
    return "ACCEPTED", json.dumps(result["counts"]), time.perf_counter() - clock


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("certificate", type=Path)
    parser.add_argument("cells", type=Path)
    parser.add_argument("--module", type=Path, required=True)
    parser.add_argument("--only", default=None)
    parser.add_argument("--json", type=Path, default=None)
    args = parser.parse_args()
    v = load(args.module, "audited_kernel_verifier")
    import hashlib

    cells = v.file_cells(args.cells, hashlib.sha256(args.cells.read_bytes()).hexdigest())
    seed, node, _, _ = load_certificate(args.certificate)
    rows: list[dict[str, Any]] = []
    with tempfile.TemporaryDirectory() as scratch:
        base = Path(scratch) / "unmutated"
        save_certificate(base, seed, node)
        verdict, message, seconds = run(v, base, cells)
        rows.append({"mutation": "(unmutated)", "verdict": verdict, "message": message[:160], "seconds": round(seconds, 2)})
        print(json.dumps(rows[-1]), flush=True)
        if verdict != "ACCEPTED":
            return 2
        for name, mutation in MUTATIONS.items():
            if args.only and not any(part in name for part in args.only.split(",")):
                continue
            s2, n2 = copy.deepcopy(seed), copy.deepcopy(node)
            try:
                expected = mutation(s2, n2, v)
            except RuntimeError as why:
                rows.append({"mutation": name, "verdict": "NOT-APPLICABLE", "message": str(why)})
                print(json.dumps(rows[-1]), flush=True)
                continue
            target = Path(scratch) / name
            save_certificate(target, s2, n2)
            verdict, message, seconds = run(v, target, cells)
            if verdict == "REFUSED" and expected and expected not in message:
                verdict = "REFUSED-ELSEWHERE"
            rows.append({"mutation": name, "expected": expected, "verdict": verdict, "message": message[:160], "seconds": round(seconds, 2)})
            print(json.dumps(rows[-1]), flush=True)
    if args.json:
        args.json.write_text(json.dumps(rows, indent=1) + "\n")
    accepted = [r["mutation"] for r in rows[1:] if r["verdict"] == "ACCEPTED"]
    print(json.dumps({"mutations": len(rows) - 1, "accepted_unsound": accepted}))
    return 1 if accepted else 0


if __name__ == "__main__":
    sys.exit(main())
