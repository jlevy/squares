Run from the worktree's packing/ (cwd) with the main venv's python:

    PYTHONPATH=<wt>/packing/src:<wt>/packing:<plugin dir> python3 <this file> [--only A,B]

Each mutant is one textual patch of the worktree's verifier (64474e45), asserted to apply
exactly once, written to a scratch copy. Against each: the committed kernel tests of the
worktree's `tests/test_verify_n17_certificates.py` (P2's selection, with the mutant bound
under the committed module's name by `mutant_plugin`), and R6's doctored refined
certificates (`mutate_refined.py.txt`) on K2's blind and wall closures. CAUGHT when a test
fails, or a doctored certificate is accepted, or the sound certificate is refused.
"""

def run(command: list[str], env: dict[str, str], timeout: int = 600) -> tuple[int, str]:
    completed = subprocess.run(command, cwd=PACKING, capture_output=True, text=True, check=False, timeout=timeout, env={**os.environ, **env})
    return completed.returncode, completed.stdout + completed.stderr


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--only", default=None)
    parser.add_argument("--skip-tests", action="store_true")
    parser.add_argument("--skip-certs", action="store_true")
    parser.add_argument("--json", type=Path, default=HERE / "refinement-mutants-r6.json")
    args = parser.parse_args()
    SCRATCH.mkdir(parents=True, exist_ok=True)
    source = VERIFIER.read_text(encoding="utf-8")
    results = []
    for name, weakens, old, new in [("control-unmutated", "nothing", "    require(cursor == 1, ", "    require(cursor == 1, "), *P2_MUTANTS, *R6_MUTANTS]:
        if args.only and not any(part in name for part in args.only.split(",")):
            continue
        assert source.count(old) == 1, (name, source.count(old))
        mutant = SCRATCH / f"{name}.py"
        mutant.write_text(source.replace(old, new), encoding="utf-8")
        clock = time.perf_counter()
        record: dict[str, object] = {"mutant": name, "weakens": weakens}
        if not args.skip_tests:
            code, out = run(
                [sys.executable, "-m", "pytest", "tests/test_verify_n17_certificates.py", "-q", "-p", "no:cacheprovider", "-p", "mutant_plugin", "-m", "not slow", "-k", TESTS, "-rf", "--no-header"],
                env={"R6_MUTANT": str(mutant)},
            )
            record["tests"] = "CAUGHT" if code != 0 else "missed"
            record["tests_failed"] = [line.split("::", 1)[1].split(" - ")[0] for line in out.splitlines() if line.startswith("FAILED")]
            record["tests_summary"] = next((line.strip() for line in out.splitlines() if " passed" in line or " failed" in line), "")[:100]
        if not args.skip_certs:
            verdicts = {}
            for label, directory in CERTS.items():
                code, out = run(
                    [sys.executable, str(HERE / "mutate_refined.py.txt"), str(directory), str(directory / "cells.json"), "--module", str(mutant)],
                    env={},
                )
                lines = [json.loads(line) for line in out.splitlines() if line.startswith("{")]
                unmutated = next((line for line in lines if line.get("mutation") == "(unmutated)"), {})
                if unmutated.get("verdict") != "ACCEPTED":
                    verdicts[label] = f"CAUGHT (refuses the sound certificate: {unmutated.get('message', '')[:80]})"
                else:
                    summary = lines[-1] if lines and "accepted_unsound" in lines[-1] else {}
                    accepted = summary.get("accepted_unsound", [])
                    verdicts[label] = "CAUGHT: " + ", ".join(accepted) if accepted else "missed"
            record["certs"] = verdicts
        record["seconds"] = round(time.perf_counter() - clock, 1)
        results.append(record)
        print(json.dumps(record), flush=True)
    args.json.write_text(json.dumps(results, indent=1) + "\n")
    return 0


if __name__ == "__main__":
    sys.exit(main())
