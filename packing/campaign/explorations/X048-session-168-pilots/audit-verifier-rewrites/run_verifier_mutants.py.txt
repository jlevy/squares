"""Lane R6's mutants of the kernel verifier's new code, and what catches each.

Run from packing/:

    .venv/bin/python3 <this file> [--only NAME] [--skip-certs]

Each mutant is one textual patch of the committed verifier, asserted to apply exactly
once, written to a scratch copy. Against each mutant run: the differential harness
(`audit_kernel_rewrite`), the committed kernel tests in `tests/test_verify_n17_certificates.py`
(through `mutant_plugin`, which binds the mutant under the committed module's name), and
the certificate mutations (`mutate_cert`) on the 8-bin W7 fixture. A verifier mutant is
CAUGHT by a check when that check fails on it. Prints one JSON line per mutant and writes
`verifier-mutants.json`.
"""

import subprocess
import sys
import time
from pathlib import Path

PACKING = HERE.parents[3]
COMMITTED = PACKING / "devtools" / "verify_n17_kernel_certificate.py"
SCRATCH = Path(os.environ.get("R6_SCRATCH", "/tmp/r6-verifier-mutants"))
PYTHON = str(PACKING / ".venv" / "bin" / "python3")
KERNEL_TESTS = "kernel or facets or sweep or between or republished or facet_cache"

def run(command: list[str], env: dict[str, str] | None = None, timeout: int = 1200) -> tuple[int, str]:
    completed = subprocess.run(
        command, cwd=PACKING, capture_output=True, text=True, check=False, timeout=timeout,
        env={**os.environ, **(env or {})},
    )
    return completed.returncode, completed.stdout + completed.stderr


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--only", default=None)
    parser.add_argument("--skip-certs", action="store_true")
    parser.add_argument("--skip-tests", action="store_true")
    parser.add_argument("--skip-harness", action="store_true")
    parser.add_argument(
        "--whole-file", action="store_true", help="run every test of the file, not the -k subset"
    )
    args = parser.parse_args()
    select = [] if args.whole_file else ["-k", KERNEL_TESTS]
    SCRATCH.mkdir(parents=True, exist_ok=True)
    plugin_dir = SCRATCH / "plugin"
    plugin_dir.mkdir(exist_ok=True)
    shutil.copy(HERE / "mutant_plugin.py.txt", plugin_dir / "mutant_plugin.py")
    source = COMMITTED.read_text()
    results = []
    for name, weakens, old, new in MUTANTS:
        if args.only and not any(part in name for part in args.only.split(",")):
            continue
        assert source.count(old) == 1, (name, source.count(old))
        mutant = SCRATCH / f"{name}.py"
        mutant.write_text(source.replace(old, new))
        clock = time.perf_counter()
        record = {"mutant": name, "weakens": weakens}
        if not args.skip_harness:
            code, out = run([PYTHON, str(HERE / "audit_kernel_rewrite.py.txt"), "--module", str(mutant), "--trials", "120", "--seed", "2026"])
            record["harness"] = "CAUGHT" if code != 0 else "missed"
            record["harness_note"] = next((line for line in out.splitlines() if '"disagreements": [' in line and '"disagreements": []' not in line), "")[:200]
        if not args.skip_tests:
            code, out = run(
                [PYTHON, "-m", "pytest", "tests/test_verify_n17_certificates.py", *select, "-p", "mutant_plugin", "-p", "no:cacheprovider", "-q", "--no-header"],
                env={"R6_MUTANT": str(mutant), "PYTHONPATH": f"{plugin_dir}:{PACKING}"},
            )
            record["tests"] = "CAUGHT" if code != 0 else "missed"
            record["tests_failed"] = [line.split("::", 1)[1].split(" - ")[0] for line in out.splitlines() if line.startswith("FAILED")]
            record["tests_summary"] = next((line.strip() for line in out.splitlines() if " passed" in line or " failed" in line), "")[:120]
        if not args.skip_certs:
            code, out = run(
                [PYTHON, str(HERE / "mutate_cert.py.txt"), str(HERE / "fixture-w7-bins8"), "--module", str(mutant)],
                env={"PYTHONPATH": str(PACKING)},
            )
            lines = [json.loads(line) for line in out.splitlines() if line.startswith("{")]
            summary = lines[-1] if lines and "accepted_unsound" in lines[-1] else {}
            unmutated = next((line for line in lines if line.get("mutation") == "(unmutated)"), {})
            if unmutated.get("verdict") != "ACCEPTED":
                record["certs"] = "CAUGHT (refuses the sound fixture)"
                record["certs_note"] = unmutated.get("message", "")[:160]
            else:
                accepted = summary.get("accepted_unsound", [])
                record["certs"] = "CAUGHT" if accepted else "missed"
                record["certs_note"] = ", ".join(accepted)[:200]
        record["seconds"] = round(time.perf_counter() - clock, 1)
        results.append(record)
        print(json.dumps(record), flush=True)
    (HERE / "verifier-mutants.json").write_text(json.dumps(results, indent=1) + "\n")
    return 0


if __name__ == "__main__":
    sys.exit(main())
