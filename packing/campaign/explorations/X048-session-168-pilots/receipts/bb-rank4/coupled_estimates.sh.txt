#!/bin/bash
cd /home/user/squares/packing
export OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
P3=/tmp/claude-0/-home-user-squares/9ef40ca7-d637-5e9a-8f7f-e71a05712560/scratchpad/lanes/p2-taylor
A=interior-SW,interior-NW,interior-W,interior-S,interior-N,interior-SE
I8=corner-SE,side-N0,side-S2,interior-SW,interior-NW,interior-W,interior-S,interior-N
N8=corner-NW,corner-NE,side-N0,side-N1,side-N2,side-W2,side-E2,interior-NW
DIVES=${DIVES:-10}
for spec in "A:$A" "I8:$I8" "N8:$N8"; do
  name=${spec%%:*}; cells=${spec#*:}
  for mode in interval coupled coupled-taylor; do
    case $mode in interval) flags="--interval";; coupled) flags="";; coupled-taylor) flags="--taylor-rows";; esac
    timeout 600 .venv/bin/python3 -m devtools.prototype_n17_bb_coupled --cells $cells --dives $DIVES --seed 2 $flags --output $P3/coupled-$name-$mode.json >> $P3/coupled.log 2>&1
    echo "$name $mode exit $?" >> $P3/coupled.log
  done
done
echo DONE >> $P3/coupled.log
