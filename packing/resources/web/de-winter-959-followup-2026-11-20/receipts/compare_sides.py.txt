def header(path):
    data = path.read_text().split("  squares:", 1)[0]
    side = re.search(r"  side:\n((?:  - .+\n)+)", data).group(1)
    poly = re.search(r"    minimal_polynomial:\n((?:    - .+\n)+)", data).group(1)
    interval = re.search(r"    isolating_interval:\n((?:    - .+\n)+)", data).group(1)
    parse = lambda s: [F(line.strip()[2:].strip("'\"")) for line in s.splitlines()]
    return parse(side), list(reversed(parse(poly))), parse(interval)

def enclosure(coefficients, interval):
    lo = hi = F(0)
    for c in reversed(coefficients):
        products = [lo * interval[0], lo * interval[1], hi * interval[0], hi * interval[1]]
        lo, hi = min(products) + c, max(products) + c
    return [lo, hi]

def describe(interval):
    out = []
    for value, rounding in zip(interval, [ROUND_FLOOR, ROUND_CEILING]):
        with localcontext() as ctx:
            ctx.prec = 60
            ctx.rounding = rounding
            out.append(str(Decimal(value.numerator) / Decimal(value.denominator)))
    return {"exact": [str(x) for x in interval], "decimal_outward": out}

# Verify the advertised side polynomial at the field element modulo its field polynomial.
