"""Enumerate only sixteen predeclared one-axis clause cases with independent logic."""
import argparse
import gzip
import hashlib
import json
import os
from pathlib import Path
import resource
import sys
import tempfile
import time

SELECTED = (0, 244, 489, 734, 978, 1223, 1468, 1712,
            1957, 2202, 2446, 2691, 2936, 3180, 3425, 3670)
RAW_SHA = "d7dd3ba492405d5b9a61dbd6a18a21164dc9c18b9b1b64d6d4633de6c9f40237"
GZIP_SHA = "916f3c5692170312257d014126b6f64a4154648f0b474040ca7c771a97785808"
RAW_BYTES = 24888016
GZIP_BYTES = 785142
BUFFER_CAP = 1024*1024


class Refusal(Exception):
    pass


def require(condition, reason):
    if not condition:
        raise Refusal(reason)


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()


def sha_json(value):
    return hashlib.sha256(canonical(value)).hexdigest()


def peak_rss():
    return resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * (1 if sys.platform == "darwin" else 1024)


class Budget:
    def __init__(self, started, node_limit=1000000, model_limit=100000, per_case_limit=50000):
        self.started = started
        self.nodes = 0
        self.models = 0
        self.node_limit = node_limit
        self.model_limit = model_limit
        self.per_case_limit = per_case_limit

    def check(self):
        require(time.monotonic()-self.started < 55, "internal-wall-cap")
        require(peak_rss() <= int(0.23*2**30), "internal-worker-rss-cap")

    def visit(self):
        self.nodes += 1
        require(self.nodes <= self.node_limit, "search-node-cap")
        if self.nodes % 128 == 1:
            self.check()


def verify_input(path, budget):
    raw_hash, stored_hash = hashlib.sha256(), hashlib.sha256()
    stored_bytes = 0
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(65536), b""):
            stored_bytes += len(block)
            require(stored_bytes <= 4*1024*1024, "compressed-input-size-cap")
            stored_hash.update(block)
    require(stored_bytes == GZIP_BYTES and stored_hash.hexdigest() == GZIP_SHA, "stored-input-identity")
    raw_bytes = 0
    with gzip.open(path, "rb") as stream:
        for block in iter(lambda: stream.read(65536), b""):
            raw_bytes += len(block)
            require(raw_bytes <= 32*1024*1024, "expanded-input-size-cap")
            raw_hash.update(block)
            budget.check()
    require(raw_bytes == RAW_BYTES and raw_hash.hexdigest() == RAW_SHA, "raw-input-identity")


class JsonStream:
    def __init__(self, source):
        self.source = source
        self.buffer = ""
        self.position = 0
        self.eof = False
        self.decoder = json.JSONDecoder()
        self.maximum_buffer = 0

    def compact(self):
        if self.position:
            self.buffer = self.buffer[self.position:]
            self.position = 0

    def fill(self):
        self.compact()
        if self.eof:
            return False
        available = BUFFER_CAP - len(self.buffer)
        require(available > 0, "json-parser-buffer-cap")
        more = self.source.read(min(65536, available))
        if not more:
            self.eof = True
            return False
        self.buffer += more
        require(len(self.buffer.encode("utf-8")) <= BUFFER_CAP, "json-parser-byte-buffer-cap")
        self.maximum_buffer = max(self.maximum_buffer, len(self.buffer.encode("utf-8")))
        return True

    def whitespace(self):
        while True:
            while self.position < len(self.buffer) and self.buffer[self.position].isspace():
                self.position += 1
            if self.position < len(self.buffer) or not self.fill():
                return

    def peek(self):
        self.whitespace()
        require(self.position < len(self.buffer), "unexpected-json-end")
        return self.buffer[self.position]

    def take(self, symbol):
        require(self.peek() == symbol, "json-delimiter-"+symbol)
        self.position += 1

    def value(self):
        self.whitespace()
        self.compact()
        while True:
            try:
                value, end = self.decoder.raw_decode(self.buffer)
            except json.JSONDecodeError:
                require(self.fill(), "incomplete-or-invalid-json-value")
                continue
            # A number at the boundary might be only a prefix of the next chunk.
            if end == len(self.buffer) and not self.eof:
                if self.fill():
                    continue
            raw = self.buffer[:end].encode("utf-8")
            self.position = end
            return value, raw

    def finish(self):
        self.whitespace()
        require(self.position == len(self.buffer) and self.eof, "trailing-json-data")


def parse_selected(path, budget):
    chosen = {}
    root = {}
    keys = set()
    count = 0
    with gzip.open(path, "rt", encoding="utf-8") as source:
        reader = JsonStream(source)
        reader.take("{")
        while True:
            key, unused = reader.value()
            require(isinstance(key, str) and key not in keys, "root-json-key")
            keys.add(key)
            reader.take(":")
            if key == "cases":
                reader.take("[")
                if reader.peek() != "]":
                    while True:
                        row, raw = reader.value()
                        require(len(raw) <= 256*1024, "source-case-size-cap")
                        require(type(row.get("case_index")) is int and row["case_index"] == count,
                                "case-index-order")
                        if count in SELECTED:
                            chosen[count] = (row, hashlib.sha256(raw).hexdigest())
                        count += 1
                        require(count <= 3671, "source-case-count-cap")
                        budget.check()
                        if reader.peek() == "]":
                            break
                        reader.take(",")
                reader.take("]")
            else:
                root[key], unused = reader.value()
            if reader.peek() == "}":
                break
            reader.take(",")
        reader.take("}")
        reader.finish()
        parser_maximum = reader.maximum_buffer
    require(count == 3671 and set(chosen) == set(SELECTED), "selected-case-completeness")
    require(root["format"] == "independent-one-axis-case-clauses-v1", "source-format")
    require(root["summary"]["status"] == "complete-canonical-case-to-clause-audit", "source-not-complete")
    require(root["summary"]["counts"]["canonical_cases"] == 3671
            and root["summary"]["counts"]["path_applications"] == 554159,
            "source-summary-counts")
    return chosen, {"source_cases_parsed": count, "selected_cases": list(SELECTED),
                    "maximum_json_buffer_bytes": parser_maximum,
                    "source_ordered_case_rows_sha256": root["summary"]["ordered_case_rows_sha256"]}


def validate_formula(clauses, variables):
    require(type(variables) is int and 0 <= variables <= 32, "edge-domain")
    domain = (1 << variables)-1
    for positive, negative in clauses:
        require(type(positive) is type(negative) is int
                and positive >= 0 and negative >= 0
                and (positive | negative) & ~domain == 0, "literal-domain")
        require(positive & negative == 0, "tautological-clause-not-in-domain")
    return domain


def direct_satisfied(clauses, variables, mask):
    require(type(mask) is int and 0 <= mask < 1 << variables, "model-domain")
    # Deliberately expand literals, rather than reuse solver bitwise satisfaction.
    for positive, negative in clauses:
        literals = [(i, 1) for i in range(variables) if positive >> i & 1]
        literals += [(i, 0) for i in range(variables) if negative >> i & 1]
        if not any((mask >> i & 1) == value for i, value in literals):
            return False
    return True


def compare_models(actual, expected, variables):
    require(all(type(m) is int and 0 <= m < 1 << variables for m in actual), "model-domain")
    require(len(actual) == len(set(actual)), "duplicate-model")
    require(set(actual) == set(expected), "model-set-disagreement")


class Solver:
    def __init__(self, clauses, variables, budget):
        self.clauses = tuple(clauses)
        self.variables = variables
        self.domain = validate_formula(self.clauses, variables)
        self.budget = budget
        self.models = []
        self.seen = set()
        self.conflicts = 0
        self.unit_assignments = 0
        self.satisfied_cubes = 0
        self.maximum_free_bits = 0

    def emit(self, mask):
        require(type(mask) is int and mask & ~self.domain == 0 and mask >= 0, "model-domain")
        require(mask not in self.seen, "duplicate-model")
        require(all((mask & p) or ((~mask) & n) for p, n in self.clauses),
                "emitted-model-fails-original-clause")
        require(self.budget.models < self.budget.model_limit, "total-model-cap")
        require(len(self.models) < self.budget.per_case_limit, "case-model-cap")
        self.budget.models += 1
        self.seen.add(mask)
        self.models.append(mask)
        if self.budget.models % 128 == 1:
            self.budget.check()

    def propagate(self, assigned, ones):
        while True:
            force_zero = force_one = 0
            all_satisfied = True
            for positive, negative in self.clauses:
                if (positive & ones) or (negative & assigned & ~ones):
                    continue
                all_satisfied = False
                free = (positive | negative) & ~assigned
                if not free:
                    self.conflicts += 1
                    return None
                if free & (free-1) == 0:
                    if free & positive:
                        force_one |= free
                    else:
                        force_zero |= free
            if force_zero & force_one:
                self.conflicts += 1
                return None
            forced = force_zero | force_one
            if not forced:
                return assigned, ones, all_satisfied
            require(forced & assigned == 0, "unit-reassigns-known-variable")
            self.unit_assignments += forced.bit_count()
            assigned |= forced
            ones |= force_one

    def search(self, assigned=0, ones=0):
        self.budget.visit()
        state = self.propagate(assigned, ones)
        if state is None:
            return
        assigned, ones, all_satisfied = state
        unassigned = self.domain ^ assigned
        if all_satisfied:
            self.satisfied_cubes += 1
            self.maximum_free_bits = max(self.maximum_free_bits, unassigned.bit_count())
            subset = unassigned
            while True:
                self.emit(ones | subset)
                if subset == 0:
                    return
                subset = (subset-1) & unassigned
        branch = unassigned & -unassigned
        require(branch != 0, "unsatisfied-complete-assignment")
        self.search(assigned | branch, ones)
        self.search(assigned | branch, ones | branch)


FIXTURES = (
    ("no-clauses", (), 8),
    ("positive-unit", ((1, 0),), 4),
    ("negative-unit", ((0, 1),), 4),
    ("contradictory-units", ((1, 0), (0, 1)), 0),
    ("empty-clause", ((0, 0),), 0),
    ("positive-two-literal", ((3, 0),), 6),
    ("exclusive-or", ((3, 0), (0, 3)), 4),
    ("two-signed-units", ((1, 0), (0, 2)), 2),
    ("duplicate-clause", ((3, 0), (3, 0)), 6),
    ("subsumed-clause", ((1, 0), (3, 0)), 4),
    ("equivalence", ((1, 2), (2, 1)), 4),
    ("unit-chain", ((1, 0), (2, 1), (4, 2)), 1),
)


def run_controls(started, output):
    for name, clauses, expected_count in FIXTURES:
        expected = [mask for mask in range(8) if direct_satisfied(clauses, 3, mask)]
        require(len(expected) == expected_count, "hand-count-fixture-"+name)
        solver = Solver(clauses, 3, Budget(started))
        solver.search()
        compare_models(solver.models, expected, 3)
        output.append({"control": name, "expected": "accept", "expected_models": expected,
                       "actual_models": sorted(solver.models), "passed": True})

    def refusal(name, operation, reason):
        try:
            operation()
        except Refusal as error:
            require(str(error) == reason, "wrong-control-refusal-"+name)
            output.append({"control": name, "expected": "refuse", "refusal": reason, "passed": True})
        else:
            raise Refusal("control-falsely-accepted-"+name)

    wrong = Solver(((0, 1),), 3, Budget(started))
    wrong.search()
    expected_positive = [1, 3, 5, 7]
    refusal("reversed-polarity", lambda: compare_models(wrong.models, expected_positive, 3),
            "model-set-disagreement")
    refusal("skipped-child", lambda: compare_models([0, 2, 4, 6], list(range(8)), 3),
            "model-set-disagreement")
    refusal("duplicate-emission", lambda: compare_models([*range(8), 0], list(range(8)), 3),
            "duplicate-model")
    refusal("outside-edge-domain", lambda: compare_models([8], [], 3), "model-domain")
    limited = Solver((), 3, Budget(started, model_limit=1))
    refusal("forced-emission-cap", limited.search, "total-model-cap")
    require(len(output) == 17, "control-count")


def selected_formula(row):
    variables = row["edge_count"]
    require(type(variables) is int and 1 <= variables <= 32, "case-edge-count")
    clauses = []
    origins = []
    for item in row["clauses"]:
        positive, negative = item["positive_mask"], item["negative_mask"]
        require(bool(positive) != bool(negative), "source-pure-clause-shape")
        require(item["path_origins"], "source-empty-origin-list")
        origins.extend(item["path_origins"])
        clauses.append((positive, negative))
    validate_formula(clauses, variables)
    require(len(clauses) == len(set(clauses)), "source-duplicate-clause")
    require(len(origins) == len(set(origins)) == row["excluded_paths"], "source-exclusion-origins")
    require(all(type(i) is int and 0 <= i < len(row["path_profile_ids"]) for i in origins),
            "source-path-origin-domain")
    require(row["excluded_paths"] + row["not_certified_paths"] == len(row["path_profile_ids"]),
            "source-path-partition")
    return clauses, variables


def retain(path, result):
    require(not path.exists(), "output-already-exists")
    temporary = None
    raw_hash = hashlib.sha256()
    raw_bytes = 0
    try:
        with tempfile.NamedTemporaryFile(mode="w+b", dir=path.parent, prefix=path.name+".",
                                         suffix=".partial", delete=False) as stream:
            temporary = stream.name
            with gzip.GzipFile(filename="", mode="wb", fileobj=stream, mtime=0, compresslevel=6) as output:
                for piece in json.JSONEncoder(sort_keys=True, separators=(",", ":"), allow_nan=False).iterencode(result):
                    data = piece.encode()
                    raw_bytes += len(data)
                    require(raw_bytes+1 <= 16*1024*1024, "raw-output-size-cap")
                    raw_hash.update(data)
                    output.write(data)
                output.write(b"\n")
                raw_hash.update(b"\n")
                raw_bytes += 1
            stream.flush()
            os.fsync(stream.fileno())
            require(stream.tell() <= 4*1024*1024, "compressed-output-size-cap")
        os.replace(temporary, path)
        temporary = None
    finally:
        if temporary:
            Path(temporary).unlink(missing_ok=True)
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(65536), b""):
            h.update(block)
    replay_hash = hashlib.sha256()
    replay_bytes = 0
    with gzip.open(path, "rb") as stream:
        for block in iter(lambda: stream.read(65536), b""):
            replay_bytes += len(block)
            require(replay_bytes <= 16*1024*1024, "output-roundtrip-size-cap")
            replay_hash.update(block)
    require(replay_bytes == raw_bytes and replay_hash.hexdigest() == raw_hash.hexdigest(),
            "output-roundtrip-identity")
    return {"raw_bytes": raw_bytes, "raw_sha256": raw_hash.hexdigest(),
            "stored_bytes": path.stat().st_size, "stored_sha256": h.hexdigest(),
            "roundtrip_verified": True}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    require(not args.output.exists() and not Path(str(args.output)+".identity.json").exists(), "output-already-exists")
    started, cpu_started = time.monotonic(), time.process_time()
    budget = Budget(started)
    result = {
        "format": "independent-one-axis-graph-pilot-v1", "status": "incomplete",
        "scope": "Only the sixteen predeclared canonical cases. No complete 3671-case graph census, producer-total verification, graph-to-bank coverage or theorem promotion.",
        "input_raw_sha256": RAW_SHA, "input_gzip_sha256": GZIP_SHA,
        "selected_case_indices": list(SELECTED),
        "checker_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "exposure": "Read producer solver previously; independent first-party enumeration, with accepted exact clause data only. No producer imports.",
        "controls": [], "cases": [], "phase": "input-verification",
    }
    current_solver = None
    current_row = None
    try:
        require(SELECTED == tuple(i*3670//15 for i in range(16)), "selection-formula")
        verify_input(args.input, budget)
        result["phase"] = "streaming-case-selection"
        selected, selection_receipt = parse_selected(args.input, budget)
        result["selection_receipt"] = selection_receipt
        result["phase"] = "controls"
        run_controls(started, result["controls"])
        for case_index in SELECTED:
            budget.check()
            result["phase"] = "graph-enumeration"
            row, input_row_sha = selected[case_index]
            clauses, variables = selected_formula(row)
            current_row = {"case_index": case_index, "status": "incomplete",
                           "vacancy": row["vacancy"], "sign_mask": row["sign_mask"],
                           "aligned_index": row["aligned_index"], "edge_count": variables,
                           "clause_count": len(clauses), "input_row_sha256": input_row_sha,
                           "ordered_clause_pairs_sha256": sha_json(clauses), "models": []}
            result["cases"].append(current_row)
            current_solver = Solver(clauses, variables, budget)
            current_row["models"] = current_solver.models
            before_nodes = budget.nodes
            case_started, case_cpu = time.monotonic(), time.process_time()
            current_solver.search()
            # Separate literal-by-literal audit of every retained model.
            for model_index, mask in enumerate(current_solver.models):
                require(direct_satisfied(clauses, variables, mask), "direct-literal-model-audit")
                if model_index % 128 == 0:
                    budget.check()
            current_solver.models.sort()
            current_row.update({"status": "complete", "model_count": len(current_solver.models),
                                "search_nodes": budget.nodes-before_nodes,
                                "conflicts": current_solver.conflicts,
                                "unit_assignments": current_solver.unit_assignments,
                                "satisfied_cubes": current_solver.satisfied_cubes,
                                "maximum_free_bits": current_solver.maximum_free_bits,
                                "wall_seconds": time.monotonic()-case_started,
                                "cpu_seconds": time.process_time()-case_cpu,
                                "ordered_models_sha256": sha_json(current_solver.models)})
            current_solver = None
            current_row = None
        require(len(result["cases"]) == 16 and all(row["status"] == "complete" for row in result["cases"]),
                "pilot-case-completeness")
        budget.check()
        result["status"] = "complete-sixteen-case-graph-pilot"
        result["phase"] = "complete"
    except Exception as error:
        result["stop_reason"] = {"type": type(error).__name__, "reason": str(error)}
        if current_solver is not None and current_row is not None:
            current_row["model_count_before_stop"] = len(current_solver.models)
    result["counts"] = {"selected_cases": 16, "complete_cases": sum(row["status"] == "complete" for row in result["cases"]),
                        "retained_models": sum(len(row.get("models", [])) for row in result["cases"]),
                        "search_nodes": budget.nodes, "passed_controls": len(result["controls"])}
    result["worker_wall_seconds_before_output"] = time.monotonic()-started
    result["worker_cpu_seconds_before_output"] = time.process_time()-cpu_started
    result["peak_rss_bytes_before_output"] = peak_rss()
    identity = retain(args.output, result)
    summary = {"status": result["status"], "counts": result["counts"],
               "stop_reason": result.get("stop_reason"), "receipt": identity,
               "worker_wall_seconds": time.monotonic()-started,
               "worker_cpu_seconds": time.process_time()-cpu_started, "peak_rss_bytes": peak_rss()}
    sidecar = Path(str(args.output)+".identity.json")
    with sidecar.open("x") as stream:
        json.dump(summary, stream, indent=2)
        stream.write("\n")
    print(json.dumps(summary))
    return 0 if result["status"] == "complete-sixteen-case-graph-pilot" else 1


if __name__ == "__main__":
    raise SystemExit(main())
