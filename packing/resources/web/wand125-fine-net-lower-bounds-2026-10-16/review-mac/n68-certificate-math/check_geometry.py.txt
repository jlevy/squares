"""Analytical area fixtures for the reviewer's clipping implementation."""
import argparse
import json
from fractions import Fraction as F
from pathlib import Path

from exact_geometry import Measure, capture_at, require

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--output', type=Path, required=True)
args = parser.parse_args()
measure = Measure(5, F(2), F(1, 2), F(1, 10), 6, F(2), 1,
                  {(F(0), F(1), F(0), F(2)): F(1)})
checks = []
for index in (0, 1, 5):
    for pose, expected in [((F(1, 2), F(1)), F(1, 4)),
                           ((F(1), F(1)), F(1, 8)),
                           ((F(3, 2), F(1)), F(0)),
                           ((F(0), F(0)), F(1, 16))]:
        observed = capture_at(measure, pose, index)
        require(observed == expected, 'analytical geometry fixture mismatch')
        checks.append({'r': index, 'pose': list(map(str, pose)), 'capture': str(observed)})
args.output.write_text(json.dumps({'status': 'PASS', 'analytical_fixtures': checks}, indent=2)+'\n')
print('12 exact analytical clipping checks passed')
